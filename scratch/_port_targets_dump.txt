  10| def anchor_plain(text):

  11|     import unicodedata

  12|     text = unicodedata.normalize('NFD', normalize_text(text).lower())

  13|     text = ''.join(ch for ch in text if unicodedata.category(ch) != 'Mn')

  14|     return text.replace('\u0111', 'd')

  15| 

  16| def _append_unique(values, value):

  17|     value = normalize_text(value)

  18|     if value and value not in values:

  19|         values.append(value)

  20| 

  21| def _plain_has_any(plain, needles):

  22|     return any(needle in plain for needle in needles)

 126| def is_generic_atomic_question(text, domain_profile=None):

 127|     plain = anchor_plain(text)

 128|     generic_markers = [

 129|         'tai lieu', 'chung cu', 'ho so', 'don yeu cau', 'hop dong', 'tranh chap', 'xu ly'

 130|     ]

 131|     if not _plain_has_any(plain, generic_markers):

 132|         return False

 133|     domain_profile = domain_profile or {}

 134|     anchors = domain_profile.get('anchor_terms', [])

 135|     return not any(anchor_plain(anchor) in plain for anchor in anchors)

 136| 

 137| def repair_generic_atomic_question(text, domain_profile):

 138|     text = normalize_text(text)

 139|     if not is_generic_atomic_question(text, domain_profile):

 140|         return text

 141|     anchors = [a for a in domain_profile.get('anchor_terms', []) if normalize_text(a)]

 142|     if not anchors:

 143|         return text

 144|     suffix = ' về ' + ', '.join(anchors[:3])

 145|     if anchor_plain(suffix) in anchor_plain(text):

 146|         return text

 147|     if text.endswith('?'):

 148|         return text[:-1].rstrip() + suffix + '?'

 149|     return text + suffix

 151| def lexical_jaccard(a, b):

 152|     aa = set(tokenize_lexical(a, keep_stopwords=False))

 153|     bb = set(tokenize_lexical(b, keep_stopwords=False))

 154|     if not aa or not bb:

 155|         return 0.0

 156|     return len(aa & bb) / max(len(aa | bb), 1)

 157| 

 158| def atomic_questions_too_similar(items):

 159|     if len(items) < 2:

 160|         return False

 161|     pairs = []

 162|     for i in range(len(items)):

 163|         for j in range(i + 1, len(items)):

 164|             pairs.append(lexical_jaccard(items[i], items[j]))

 165|     return bool(pairs) and max(pairs) >= 0.62

 166| 

 167| def dedupe_atomic_questions(items, max_items=None):

 168|     out = []

 169|     for item in items:

 170|         item = normalize_text(item)

 171|         if not item:

 172|             continue

 173|         if any(normalize_key(item) == normalize_key(x) or lexical_jaccard(item, x) >= 0.78 for x in out):

 174|             continue

 175|         out.append(item)

 176|         if max_items and len(out) >= max_items:

 177|             break

 178|     return out

 211| def weighted_rrf_fuse(weighted_rankings, k=60, topk=96):

 212|     scores = defaultdict(float)

 213|     payload = {}

 214|     sources = defaultdict(list)

 215|     for ranking, weight, label in weighted_rankings:

 216|         for rank, item in enumerate(ranking, start=1):

 217|             row_idx = int(item['row_idx'])

 218|             scores[row_idx] += float(weight) / (k + rank)

 219|             payload.setdefault(row_idx, item)

 220|             sources[row_idx].append({'source': label, 'rank': rank, 'weight': float(weight)})

 221|     fused = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:topk]

 222|     out = []

 223|     for row_idx, score in fused:

 224|         item = dict(payload[row_idx])

 225|         item['rrf_score'] = float(score)

 226|         item['source_hits'] = sources[row_idx][:12]

 227|         out.append(item)

 228|     return out

 360| def article_bounds_for_complexity(complexity):

 361|     """Per-tier article quota: {min_k, target_k, max_k}.

 362| 

 363|     simple  -> 1 article,   complex -> up to 12,  medium -> up to 4.

 364|     Each max_k is capped by SUBMIT_ARTICLE_MAX (the hard ceiling).

 365|     """

 366|     if complexity == 'simple':

 367|         return {'min_k': 1, 'target_k': 1, 'max_k': min(SUBMIT_ARTICLE_MAX_SIMPLE, SUBMIT_ARTICLE_MAX)}

 368|     if complexity == 'complex':

 369|         return {'min_k': 4, 'target_k': 10, 'max_k': min(SUBMIT_ARTICLE_MAX_COMPLEX, SUBMIT_ARTICLE_MAX)}

 370|     return {'min_k': 2, 'target_k': 4, 'max_k': min(SUBMIT_ARTICLE_MAX_MEDIUM, SUBMIT_ARTICLE_MAX)}
